  result <- data|>
    filter(!is.na(GLA.Population.Estimate.2016)) |>
    mutate(popul_estim =  gsub("£|,", "",GLA.Population.Estimate.2016)) |>
    mutate(household_estim = gsub("£|,", "",GLA.Household.Estimate.2016)) |>
    mutate(household_density = as.numeric( popul_estim) /as.numeric (household_estim) )|>
    select(Area.name, household_density) |>
    filter(Area.name == area_name)
    return(result)
